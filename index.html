using UnityEngine; using UnityEngine.SceneManagement; public class TygoSimpleGame : MonoBehaviour { public float speed = 6f; public float jumpForce = 8f; public int lives = 3; private Rigidbody2D rb; private bool grounded; void Start() { rb = GetComponent(); } void Update() { // LINKS / RECHTS float x = Input.GetAxisRaw("Horizontal"); rb.linearVelocity = new Vector2(x * speed, rb.linearVelocity.y); // SPRINGEN if (Input.GetKeyDown(KeyCode.Space) && grounded) { rb.linearVelocity = new Vector2(rb.linearVelocity.x, jumpForce); } } void TakeDamage() { lives--; if (lives <= 0) { SceneManager.LoadScene(SceneManager.GetActiveScene().buildIndex); } else { transform.position = new Vector3(0, 2, 0); } } void OnCollisionEnter2D(Collision2D col) { if (col.gameObject.CompareTag("Ground")) { grounded = true; } if (col.gameObject.CompareTag("Enemy")) { TakeDamage(); } if (col.gameObject.CompareTag("Death")) { TakeDamage(); } } void OnCollisionExit2D(Collision2D col) { if (col.gameObject.CompareTag("Ground")) { grounded = false; } } void OnTriggerEnter2D(Collider2D col) { if (col.gameObject.CompareTag("Finish")) { SceneManager.LoadScene(SceneManager.GetActiveScene().buildIndex + 1); } if (col.gameObject.CompareTag("Crystal")) { Destroy(col.gameObject); } } }
