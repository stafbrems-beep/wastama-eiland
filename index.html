// ===== PLAYER ===== let playerSpeed = 3; let playerHP = 3; // ===== MOVEMENT ===== function updatePlayer(){ if(keys["w"]){ player.position.x += Math.sin(yaw) * playerSpeed; player.position.z -= Math.cos(yaw) * playerSpeed; } if(keys["s"]){ player.position.x -= Math.sin(yaw) * playerSpeed; player.position.z += Math.cos(yaw) * playerSpeed; } if(keys["a"]){ player.position.x -= Math.cos(yaw) * playerSpeed; player.position.z -= Math.sin(yaw) * playerSpeed; } if(keys["d"]){ player.position.x += Math.cos(yaw) * playerSpeed; player.position.z += Math.sin(yaw) * playerSpeed; } } // ===== CAMERA ===== function updateCamera(){ camera.position.x = player.position.x - Math.sin(yaw) * 40; camera.position.y = player.position.y + 25; camera.position.z = player.position.z + Math.cos(yaw) * 40; camera.lookAt(player.position); } // ===== BULLETS ===== function updateBullets(){ for(let i = bullets.length - 1; i >= 0; i--){ let bullet = bullets[i]; bullet.position.add( bullet.userData.dir.clone().multiplyScalar(8) ); if(bullet.position.length() > 2000){ scene.remove(bullet); bullets.splice(i,1); } } } // ===== ENEMY AI ===== function updateEnemies(){ for(let enemy of enemies){ let dir = player.position.clone() .sub(enemy.position) .normalize(); enemy.position.add( dir.multiplyScalar(0.4) ); if(Math.random() < 0.005){ shootBot(enemy); } } } // ===== COLLISIONS ===== function checkHits(){ for(let b = bullets.length-1; b >= 0; b--){ for(let e = enemies.length-1; e >= 0; e--){ let bullet = bullets[b]; let enemy = enemies[e]; if( bullet.position.distanceTo( enemy.position ) < 8 ){ scene.remove(enemy); scene.remove(bullet); enemies.splice(e,1); bullets.splice(b,1); kills++; document.getElementById("kills").innerText = kills; // coin drop let coin = new THREE.Mesh( new THREE.CylinderGeometry(2,2,1), new THREE.MeshBasicMaterial({ color:0xffff00 }) ); coin.rotation.x = Math.PI/2; coin.position.copy(enemy.position); scene.add(coin); coinObjects.push(coin); break; } } } } // ===== COINS ===== function updateCoins(){ for(let i = coinObjects.length-1; i >= 0; i--){ if( player.position.distanceTo( coinObjects[i].position ) < 10 ){ coins++; document.getElementById("coins").innerText = coins; scene.remove(coinObjects[i]); coinObjects.splice(i,1); } } } // ===== HEALTH ===== function spawnHealthPack(){ let med = new THREE.Mesh( new THREE.BoxGeometry(5,5,5), new THREE.MeshBasicMaterial({ color:0xff00ff }) ); med.position.set( (Math.random()-0.5)*500, 3, (Math.random()-0.5)*500 ); scene.add(med); healthPacks.push(med); } setInterval(()=>{ if(gameActive){ spawnHealthPack(); } },30000); // ===== GAME LOOP ===== function updateGame(){ updatePlayer(); updateCamera(); updateBullets(); updateEnemies(); checkHits(); updateCoins(); }
